# --- PyTorch (CPU build) ---

# macOS: the normal PyPI wheel (Apple Silicon / Intel).
--extra-index-url https://download.pytorch.org/whl/cpu

numpy==1.26.4 ; sys_platform == "darwin"
numpy         ; sys_platform != "darwin"
